local function SafeParent(gui)
    if not gui then return end
    pcall(function() gui.Parent = HIDDEN end)
    if not gui.Parent then pcall(function() gui.Parent = CoreGui end) end
    if not gui.Parent then pcall(function() gui.Parent = LP:WaitForChild("PlayerGui", 3) end) end
end

-- ═══════════════ HELPERS ═══════════════
local function mk(cls, parent, props)
    local i = Instance_new(cls)
    if props then
        for k, v in pairs(props) do
            pcall(function() i[k] = v end)
        end
    end
    if parent then pcall(function() i.Parent = parent end) end
    return i
end

local function corner(p, r)
    local c = Instance_new("UICorner")
    c.CornerRadius = UDim_new(0, r or 8)
    c.Parent = p
    return c
end

local function stroke(p, col, thick, trans)
    local s = Instance_new("UIStroke")
    s.Color = col or Color3_RGB(0, 220, 255)
    s.Thickness = thick or 1
    s.Transparency = trans or 0.4
    s.Parent = p
    return s
end

local function UISlider(parent, text, minV, maxV, defaultV, cb)
    local c = mk("Frame", parent, {
        Size = UDim2_new(1, 0, 0, 40),
        BackgroundColor3 = Color3_RGB(26, 26, 36),
        BackgroundTransparency = 0.25,
        BorderSizePixel = 0,
        ZIndex = 11,
    })
    corner(c, 6)
    stroke(c, Color3_RGB(0, 220, 255), 1, 0.6)

    mk("TextLabel", c, {
        Size = UDim2_new(1, -60, 0, 14),
        Position = UDim2_new(0, 8, 0, 3),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = Color3_RGB(240, 240, 250),
        Font = Enum.Font.GothamBold,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 12,
    })
    local valLbl = mk("TextLabel", c, {
        Size = UDim2_new(0, 50, 0, 14),
        Position = UDim2_new(1, -58, 0, 3),
        BackgroundTransparency = 1,
        Text = tostring(defaultV),
        Font = Enum.Font.GothamBlack,
        TextSize = 11,
        TextColor3 = Color3_RGB(0, 220, 255),
        TextXAlignment = Enum.TextXAlignment.Right,
        ZIndex = 12,
    })

    local track = mk("Frame", c, {
        Size = UDim2_new(1, -16, 0, 5),
        Position = UDim2_new(0, 8, 0, 26),
        BackgroundColor3 = Color3_RGB(40, 40, 52),
        BorderSizePixel = 0,
        ZIndex = 12,
    })
    corner(track, 3)

    local fill = mk("Frame", track, {
        Size = UDim2_new(math_clamp((defaultV - minV) / (maxV - minV), 0, 1), 0, 1, 0),
        BackgroundColor3 = Color3_RGB(0, 220, 255),
        BorderSizePixel = 0,
        ZIndex = 13,
    })
    corner(fill, 3)

    local dragging = false
    local function upd(inp)
        if track.AbsoluteSize.X <= 0 then return end
        local ratio = math_clamp((inp.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2_new(ratio, 0, 1, 0)
        local v = math_floor(minV + (maxV - minV) * ratio)
        valLbl.Text = tostring(v)
        pcall(cb, v)
    end

    track.InputBegan:Connect(function(inp)
        if inp.UserInputType == UserInputType.MouseButton1 or inp.UserInputType == UserInputType.Touch then
            dragging = true
            upd(inp)
        end
    end)
    track.InputEnded:Connect(function() dragging = false end)
    UIS.InputChanged:Connect(function(inp)
        if dragging and (inp.UserInputType == UserInputType.MouseMovement or inp.UserInputType == UserInputType.Touch) then
            upd(inp)
        end
    end)
    return c
end

    while InfoPanel.Parent do
        RunService.RenderStepped:Wait()
        fpsCount = fpsCount + 1
        local now = tick()
        if now - fpsLast >= 1 then
            S.FPS = fpsCount
            fpsCount = 0
            fpsLast = now
        end
    end
end)

local STEAL_KW = { "steal", "grab", "pick", "cướp", "cắp", "trộm", "nhặt", "lấy", "snatch", "trứng", "egg" }
local PLACE_KW = { "place", "submit", "deliver", "đặt", "nộp", "ổ", "nest" }

local function hasKW(s, list)
    if not s then return false end
    s = string_lower(s)
    for i = 1, #list do
        if string_find(s, list[i], 1, true) then return true end
    end
    return false
end

local function GetPromptPos(p)
    local parent = p.Parent
    if not parent then return nil end
    if parent:IsA("BasePart") then return parent.Position end
    if parent:IsA("Model") and parent.PrimaryPart then return parent.PrimaryPart.Position end
    for _, c in ipairs(parent:GetChildren()) do
        if c:IsA("BasePart") then return c.Position end
    end
    return nil
end

function ScanAll()
    S.Cache = { Steal = {}, Place = {} }
    local count = 0
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            count = count + 1
            if count % 2000 == 0 then task_wait() end
            if obj:IsA("ProximityPrompt") and obj.Enabled then
                local text = tostring(obj.ActionText or "") .. " " .. tostring(obj.ObjectText or "") .. " " .. obj.Name
                local pos = GetPromptPos(obj)
                if pos then
                    if hasKW(text, PLACE_KW) and not hasKW(text, STEAL_KW) then
                        table_insert(S.Cache.Place, { p = obj, pos = pos })
                    elseif hasKW(text, STEAL_KW) then
                        table_insert(S.Cache.Steal, { p = obj, pos = pos })
                    end
                end
            end
        end
    end)
    print("[SCAN] Steal: " .. #S.Cache.Steal .. " | Place: " .. #S.Cache.Place)
end

-- Detect đang mang trứng (multi-source)
local function DetectCarrying()
    UpdateChar()
    if not Char then return false end

    -- 1) Con của nhân vật
    for _, c in ipairs(Char:GetChildren()) do
        local n = string_lower(c.Name)
        if string_find(n, "egg", 1, true) or string_find(n, "trứng", 1, true) then
            if not string_find(n, "bat", 1, true) and not string_find(n, "gậy", 1, true) then
                if c:IsA("BasePart") or c:IsA("Model") or c:IsA("MeshPart") or c:IsA("Tool") or c:IsA("Accessory") then
                    return true
                end
            end
        end
    end

    -- 2) Backpack
    local bp = LP:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            local n = string_lower(t.Name)
            if string_find(n, "egg", 1, true) or string_find(n, "trứng", 1, true) then
                return true
            end
        end
    end

    -- 3) PlayerGui — tìm text UI "thả trứng"
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then
        for _, g in ipairs(pg:GetDescendants()) do
            if (g:IsA("TextButton") or g:IsA("TextLabel")) and g.Visible then
                local t = string_lower(tostring(g.Text))
                if string_find(t, "thả trứng", 1, true) or string_find(t, "drop egg", 1, true)
                    or string_find(t, "đặt trứng", 1, true) or string_find(t, "deliver", 1, true)
                    or string_find(t, "giao trứng", 1, true) then
                    return true
                end
            end
        end
    end

    -- 4) Character children attributes
    for _, c in ipairs(Char:GetChildren()) do
        if c:IsA("Tool") then return true end
    end

    return false
end

-- WalkTo (giữ nguyên)
function WalkTo(targetPos, timeout)
    UpdateChar()
    if not Hum or not HRP then return false end
    pcall(function() Hum:MoveTo(targetPos) end)
    local start = tick()
    timeout = timeout or 15
    local lastJump = tick()
    while tick() - start < timeout do
        UpdateChar()
        if not HRP or not Hum then return false end
        if (HRP.Position - targetPos).Magnitude < 6 then return true end
        if tick() - lastJump > 2 then
            pcall(function() Hum.Jump = true end)
            pcall(function() Hum:MoveTo(targetPos) end)
            lastJump = tick()
        end
        task_wait(0.3)
    end
    return false
end

-- ========= FIX #2: SMOOTH TELEPORT =========
-- Không set vị trí từng bước cứng ngắc → dùng body velocity
function SmoothTeleTo(targetPos, timeout)
    UpdateChar()
    if not HRP or not Hum then return false end

    local dist = (HRP.Position - targetPos).Magnitude
    if dist < 3 then return true end

    local spd = S.TeleSpeed or 80
    timeout = timeout or math_min(dist / spd + 5, 20)

    -- Bật noclip tạm
    local parts = {}
    for _, p in ipairs(Char:GetDescendants()) do
        if p:IsA("BasePart") then
            parts[p] = p.CanCollide
            p.CanCollide = false
        end
    end

    local bv = Instance_new("BodyVelocity")
    bv.MaxForce = Vector3_new(1e5, 1e5, 1e5)
    bv.P = 1e4
    bv.Parent = HRP

    local start = tick()
    local ok = false
    while tick() - start < timeout do
        UpdateChar()
        if not HRP then break end
        local d = targetPos - HRP.Position
        if d.Magnitude < 3 then
            ok = true
            break
        end
        local dir = d.Unit
        local desired = dir * math_min(spd, d.Magnitude * 8)
        bv.Velocity = desired
        HRP.CFrame = CFrame_new(HRP.Position, HRP.Position + Vector3_new(dir.X, 0, dir.Z))
        RunService.RenderStepped:Wait()
    end

    pcall(function() bv:Destroy() end)

    -- Khôi phục collision
    for p, canCol in pairs(parts) do
        pcall(function()
            if p.Parent then p.CanCollide = canCol end
        end)
    end

    return ok
end

-- Alias cũ để các chỗ khác vẫn hoạt động
function TeleTo(targetPos, timeout)
    return SmoothTeleTo(targetPos, timeout)
end

-- Try steal — fix prompt hold
function TrySteal(prompt, pos)
    if not prompt or not prompt.Parent or not prompt.Enabled then return false end
    UpdateChar()
    if not HRP then return false end

    if (HRP.Position - pos).Magnitude > 12 then
        if S.StealMode == "tele" or S.StealMode == "instant" then
            SmoothTeleTo(pos, 8)
        else
            WalkTo(pos, 8)
        end
    end

    task_wait(0.15)

    local ok = pcall(function() prompt:InputHoldBegin() end)
    if ok then
        local hold = math_clamp(prompt.HoldDuration or 0.5, 0.1, 3)
        task_wait(hold)
        pcall(function() prompt:InputHoldEnd() end)
        task_wait(0.25)
    end
    return true
end

-- Try place
function TryPlace(nestPos)
    UpdateChar()
    if not HRP then return false end

    local best, bestDist = nil, 15
    for _, d in ipairs(S.Cache.Place) do
        if d.p and d.p.Parent then
            local dist = (nestPos - d.pos).Magnitude
            if dist < bestDist then
                bestDist = dist
                best = d
            end
        end
    end

    if best then
        SmoothTeleTo(best.pos, 5)
        task_wait(0.2)
        local ok = pcall(function() best.p:InputHoldBegin() end)
        if ok then
            task_wait(math_clamp(best.p.HoldDuration or 0.5, 0.1, 2))
            pcall(function() best.p:InputHoldEnd() end)
            task_wait(0.35)
        end
        return true
    end
    return false
end

-- Action tele steal
function ActionTeleSteal()
    UpdateChar()
    if not HRP then
        Notify("error", "Chưa có nhân vật", "", 2)
        return
    end
    if #S.Cache.Steal == 0 then ScanAll() end
    if #S.Cache.Steal == 0 then
        Notify("warn", "Không tìm thấy trứng", "", 2)
        return
    end
    local nearest, minDist = nil, math_huge
    for _, d in ipairs(S.Cache.Steal) do
        if d.p and d.p.Parent then
            local dist = (HRP.Position - d.pos).Magnitude
            if dist < minDist then
                minDist = dist
                nearest = d
            end
        end
    end
    if not nearest then
        Notify("warn", "Không có trứng", "", 2)
        return
    end
    SetStatus("🌀 Đang đến trứng...")
    SmoothTeleTo(nearest.pos, 10)
    task_wait(0.2)
    TrySteal(nearest.p, nearest.pos)
    S.EggsStolen = S.EggsStolen + 1
    SetStatus("✅ Đã lấy trứng")
    Notify("success", "🥚 Lấy OK!", "#" .. S.EggsStolen, 2)
end

-- ═══════════════ FIX #3: FLY ENGINE MƯỢT ═══════════════
local flyConn = nil
local flyVelocity = Vector3_new(0, 0, 0)
local flyY = 0
local flyBV = nil
local flyBG = nil

local function CreateFlyBodies()
    if not HRP then return end
    if flyBV and flyBV.Parent then return end
    flyBV = Instance_new("BodyVelocity")
    flyBV.MaxForce = Vector3_new(1e6, 1e6, 1e6)
    flyBV.P = 12500
    flyBV.Velocity = Vector3_new(0, 0, 0)
    flyBV.Parent = HRP

    flyBG = Instance_new("BodyGyro")
    flyBG.MaxTorque = Vector3_new(1e6, 1e6, 1e6)
    flyBG.P = 10000
    flyBG.D = 500
    flyBG.CFrame = HRP.CFrame
    flyBG.Parent = HRP
end

local function DestroyFlyBodies()
    if flyBV then pcall(function() flyBV:Destroy() end); flyBV = nil end
    if flyBG then pcall(function() flyBG:Destroy() end); flyBG = nil end
end

function SetFly(on)
    S.FlyMode = on
    if on then
        UpdateChar()
        if HRP then flyY = HRP.Position.Y end
        flyVelocity = Vector3_new(0, 0, 0)
        CreateFlyBodies()
        if Hum then
            pcall(function() Hum.PlatformStand = true end)
        end
        if not flyConn then
            flyConn = RunService.RenderStepped:Connect(function(dt)
                if not S.FlyMode then return end
                UpdateChar()
                if not HRP or not Hum then return end
                if Hum.Health <= 0 then
                    SetFly(false)
                    return
                end
                dt = math_min(dt, 0.1)

                local camCF = Camera.CFrame
                local camLook = camCF.LookVector
                local camRight = camCF.RightVector

                -- Làm phẳng look/right
                local flatLook = Vector3_new(camLook.X, 0, camLook.Z)
                if flatLook.Magnitude < 0.01 then flatLook = Vector3_new(0, 0, 1) end
                flatLook = flatLook.Unit

                local flatRight = Vector3_new(camRight.X, 0, camRight.Z)
                if flatRight.Magnitude < 0.01 then flatRight = Vector3_new(1, 0, 0) end
                flatRight = flatRight.Unit

                -- Input từ Humanoid.MoveDirection (joystick Roblox)
                local md = Hum.MoveDirection
                local wish = Vector3_new(0, 0, 0)
                if md.Magnitude > 0.05 then
                    wish = (flatLook * md.Z + flatRight * md.X)
                end

                -- Lên/xuống
                local upInput = 0
                if S.FlyUp then upInput = upInput + 1 end
                if S.FlyDown then upInput = upInput - 1 end

                -- Target velocity
                local targetVel = wish.Unit * S.FlySpeed
                if wish.Magnitude < 0.05 then
                    targetVel = Vector3_new(0, 0, 0)
                end
                targetVel = targetVel + Vector3_new(0, upInput * S.FlySpeed, 0)

                -- Lerp velocity với dt (mượt)
                local lerpRate = 12 * dt
                flyVelocity = flyVelocity:Lerp(targetVel, math_min(lerpRate, 1))

                -- Áp dụng lên BodyVelocity
                if flyBV and flyBV.Parent then
                    flyBV.Velocity = flyVelocity
                end

                -- BodyGyro luôn hướng theo camera
                if flyBG and flyBG.Parent then
                    flyBG.CFrame = camCF
                end
            end)
        end
        Notify("success", "✈️ Fly ON", "Dùng joystick + Space/LShift", 3)
        SetStatus("✈️ Đang bay")
    else
        if flyConn then
            flyConn:Disconnect()
            flyConn = nil
        end
        DestroyFlyBodies()
        pcall(function()
            if Hum then Hum.PlatformStand = false end
        end)
        flyVelocity = Vector3_new(0, 0, 0)
        Notify("warn", "✈️ Fly OFF", "", 2)
        SetStatus("🟢 Sẵn sàng")
    end
end

-- ═══════════════ AUTO FARM LOOP (FIX #4) ═══════════════
local FarmRunning = false

function StartFarmLoop()
    if FarmRunning then return end
    FarmRunning = true
    SetStatus("🚀 Farm bắt đầu")

    task_spawn(function()
        local lastNotifyNoNest = 0
        while S.FarmMode == "on" do
            pcall(function()
                UpdateChar()
                if not HRP or not Hum or Hum.Health <= 0 then
                    SetStatus("💀 Chết...")
                    task_wait(2)
                    return
                end

                S.IsCarrying = DetectCarrying()

                if S.IsCarrying then
                    -- ĐANG MANG TRỨNG → VỀ Ổ GIAO
                    if not S.HasNest then
                        SetStatus("⚠️ Chưa lưu ổ!")
                        if tick() - lastNotifyNoNest > 15 then
                            Notify("warn", "Chưa lưu ổ!", "Đến ổ → bấm LƯU Ổ", 4)
                            lastNotifyNoNest = tick()
                        end
                        task_wait(2)
                        return
                    end

                    SetStatus("🏠 Về ổ giao trứng...")
                    local dist = (HRP.Position - S.SafeNest).Magnitude
                    if dist > 15 then
                        SmoothTeleTo(S.SafeNest, 15)
                    elseif dist > 5 then
                        WalkTo(S.SafeNest, 5)
                    end
                    task_wait(0.3)

                    if TryPlace(S.SafeNest) then
                        task_wait(0.6)
                    end

                    task_wait(0.5)
                    S.IsCarrying = DetectCarrying()
                    if not S.IsCarrying then
                        S.EggsPlaced = S.EggsPlaced + 1
                        SetStatus("✅ Đã giao #" .. S.EggsPlaced)
                        Notify("success", "🏠 Giao OK!", "Tổng: " .. S.EggsPlaced, 2)
                    else
                        SetStatus("⚠️ Giao hụt, thử lại...")
                        task_wait(1)
                    end

                else
                    -- KHÔNG MANG TRỨNG → ĐI LẤY
                    if tick() - S.LastScan > 4 then
                        ScanAll()
                        S.LastScan = tick()
                    end

                    if #S.Cache.Steal == 0 then
                        SetStatus("🔍 Đang quét trứng...")
                        task_wait(2)
                        return
                    end

                    local nearest, minDist = nil, math_huge
                    for _, d in ipairs(S.Cache.Steal) do
                        if d.p and d.p.Parent and d.pos.Y > -50 then
                            local dist = (HRP.Position - d.pos).Magnitude
                            if dist < S.Radius and dist < minDist then
                                minDist = dist
                                nearest = d
                            end
                        end
                    end

                    if nearest then
                        SetStatus("🚶 Đến trứng (" .. math_floor(minDist) .. "m)")
                        if S.StealMode == "tele" or S.StealMode == "instant" then
                            SmoothTeleTo(nearest.pos, 15)
                        else
                            WalkTo(nearest.pos, 20)
                        end
                        task_wait(0.3)
                        SetStatus("🎯 Đang lấy trứng...")
                        TrySteal(nearest.p, nearest.pos)
                        task_wait(0.5)

                        S.IsCarrying = DetectCarrying()
                        if S.IsCarrying then
                            S.EggsStolen = S.EggsStolen + 1
                            SetStatus("✅ Lấy được #" .. S.EggsStolen)
                            Notify("success", "🥚 Lấy OK!", "#" .. S.EggsStolen, 2)
                        else
                            SetStatus("⚠️ Lấy hụt, tiếp tục...")
                        end
                    else
                        SetStatus("🔍 Không thấy trứng trong bán kính")
                        task_wait(2)
                    end
                end
            end)
            task_wait(0.3)
        end
        FarmRunning = false
        SetStatus("🟢 Sẵn sàng")
    end)
end

        S.ShowInfo = not S.ShowInfo
        InfoPanel.Visible = S.ShowInfo
    elseif inp.KeyCode == KeyCode.F5 then
        SetFly(not S.FlyMode)
    elseif inp.KeyCode == KeyCode.Space then
        if S.FlyMode then S.FlyUp = true end
    elseif inp.KeyCode == KeyCode.LeftShift then
        if S.FlyMode then S.FlyDown = true end
    end
end)
